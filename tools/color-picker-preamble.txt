useEffect(() => {
  // live demo ships with its color-picker popover open; [popover] needs showPopover()
  // and fixed top/left must be recomputed relative to the trigger (stage offset differs from live)
  const place = () => {
    document.querySelectorAll('.clp-demo [popover]').forEach((el) => {
      if (!(el instanceof HTMLElement)) return;
      const pop = el as HTMLElement;
      if (!pop.matches(':popover-open')) { try { pop.showPopover(); } catch {} }
      let trig: Element | null = pop.id ? document.querySelector(`[aria-controls="${pop.id}"]`) : null;
      if (!trig) {
        let prev = pop.previousElementSibling;
        while (prev && !trig) {
          if (prev instanceof HTMLElement && (prev.tagName === 'BUTTON' || prev.getAttribute('role') === 'button')) trig = prev;
          prev = prev.previousElementSibling;
        }
      }
      if (!trig) return;
      const tR = trig.getBoundingClientRect();
      // use layout size (offsetWidth) not getBoundingClientRect: popover's intro
      // scale animation would otherwise shrink the measured width ~4%
      const pW = pop.offsetWidth, pH = pop.offsetHeight;
      const side = pop.dataset.side || 'bottom';
      const align = pop.dataset.align || 'start';
      let top = tR.top;
      if (side === 'bottom') top = tR.bottom + 6;
      else if (side === 'top') top = tR.top - pH - 6;
      let left = tR.left;
      if (align === 'end') left = tR.right - pW;
      else if (align === 'center') left = tR.left + tR.width / 2 - pW / 2;
      pop.style.top = `${top}px`;
      pop.style.left = `${left}px`;
    });
  };
  place();
  window.addEventListener('resize', place);
  return () => window.removeEventListener('resize', place);
}, []);
